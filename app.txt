// 📍 CONFIGURAÇÃO
const SUPABASE_URL = 'https://ucrnfwoqnhurruuaxxjm.supabase.co'; // 👈 TROQUE PELA SUA URL REAL
const SUPABASE_ANON_KEY = 'sb_publishable_yAMPzra54DEUhD8evwyyNg_yP046Hwe';
const STRIPE_PUBLISHABLE_KEY = 'pk_test_51TSnqUFTSIxJv0QUZzfSnC8F6pjdqVrEgfA3pufS5JgZp02TfUiAg5AGwLXl09fehaegQardoOpfYcSzJdsQuj4j00V8L2mr5z';

const supabase = window.supabase.createClient(SUPABASE_URL, SUPABASE_ANON_KEY);
const stripe = window.Stripe(STRIPE_PUBLISHABLE_KEY);

// 🔑 FUNÇÃO UNIFICADA DE VERIFICAÇÃO DE SESSÃO
async function getSession() {
  const { data: { session }, error } = await supabase.auth.getSession();
  if (error) console.error('Erro na sessão:', error);
  return session;
}

// 🌐 LÓGICA DE INDEX.HTML (LOGIN/CADASTRO)
const emailInput = document.getElementById('email');
const passInput = document.getElementById('password');
const signupBtn = document.getElementById('signupBtn');
const toggleAuth = document.getElementById('toggleAuth');
let isSignup = true;

if (emailInput && passInput && signupBtn) {
  // Se já estiver logado, redireciona
  getSession().then(session => {
    if (session) window.location.href = 'dashboard.html';
  });

  signupBtn.addEventListener('click', async (e) => {
    e.preventDefault();
    const email = emailInput.value.trim();
    const password = passInput.value.trim();

    if (!email || !password) return alert('⚠️ Preencha e-mail e senha.');

    signupBtn.textContent = 'Processando...';
    signupBtn.disabled = true;

    try {
      if (isSignup) {
        const { error } = await supabase.auth.signUp({ email, password });
        if (error) throw error;
        alert('✅ Conta criada! Redirecionando...');
        window.location.href = 'dashboard.html';
      } else {
        const { error } = await supabase.auth.signInWithPassword({ email, password });
        if (error) throw error;
        window.location.href = 'dashboard.html';
      }
    } catch (err) {
      console.error('Auth error:', err);
      alert('❌ Erro: ' + err.message);
    } finally {
      signupBtn.textContent = isSignup ? 'Criar conta grátis' : 'Entrar';
      signupBtn.disabled = false;
    }
  });

  if (toggleAuth) {
    toggleAuth.addEventListener('click', (e) => {
      e.preventDefault();
      isSignup = !isSignup;
      signupBtn.textContent = isSignup ? 'Criar conta grátis' : 'Entrar';
      toggleAuth.textContent = isSignup ? 'Já tem conta? Entrar' : 'Não tem conta? Criar';
    });
  }
}

// 📊 LÓGICA DE DASHBOARD.HTML
document.addEventListener('DOMContentLoaded', async () => {
  const session = await getSession();

  // Proteção de rota
  if (!session) {
    console.warn('Usuário não autenticado. Redirecionando para index.');
    window.location.href = 'index.html';
    return;
  }

  // Preenche e-mail do usuário
  const emailEl = document.getElementById('userEmail');
  if (emailEl) emailEl.textContent = session.user.email;

  // Busca perfil e verifica trial
  const { data: profile } = await supabase
    .from('profiles')
    .select('trial_ends_at, subscription_status')
    .eq('id', session.user.id)
    .single();

  const now = new Date();
  const trialEnd = new Date(profile?.trial_ends_at || now);
  const isActive = trialEnd > now || profile?.subscription_status === 'active';

  const paywall = document.getElementById('paywall');
  const banner = document.getElementById('trialBanner');
  const daysEl = document.getElementById('trialDays');

  if (!isActive && paywall && banner) {
    paywall.style.display = 'flex';
    banner.style.display = 'none';
  } else if (daysEl) {
    const daysLeft = Math.max(0, Math.ceil((trialEnd - now) / (1000 * 60 * 60 * 24)));
    daysEl.textContent = daysLeft;
  }

  // Redirect pós-Stripe
  const urlParams = new URLSearchParams(window.location.search);
  if (urlParams.get('checkout') === 'success') {
    setTimeout(() => window.location.replace(window.location.pathname), 2000);
  }
});

// 💳 STRIPE CHECKOUT
const subBtn = document.getElementById('subscribeBtn');
if (subBtn) {
  subBtn.addEventListener('click', async () => {
    const session = await getSession();
    if (!session) return alert('Faça login primeiro.');

    const { error } = await stripe.redirectToCheckout({
      mode: 'subscription',
      lineItems: [{ price: 'price_1TSo5gFTSIxJv0QU4O4UQ4dr', quantity: 1 }], // 👈 SUBSTITUA PELO SEU PRICE ID
      customerEmail: session.user.email,
      subscriptionData: { metadata: { userId: session.user.id } },
      successUrl: `${location.origin}/dashboard.html?checkout=success`,
      cancelUrl: `${location.origin}/dashboard.html?checkout=cancel`
    });

    if (error) alert('Erro no checkout: ' + error.message);
  });
}

// 🚪 LOGOUT
const logoutBtn = document.getElementById('logoutBtn');
if (logoutBtn) {
  logoutBtn.addEventListener('click', async () => {
    await supabase.auth.signOut();
    window.location.href = 'index.html';
  });
}
